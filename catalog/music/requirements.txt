# HexCast - Music plugin (YouTube Music Desktop + local files)
#
# The async Socket.IO client needs aiohttp, and it is NOT pulled in by a plain
# "pip install python-socketio". Missing it looks like a working integration
# whose live feed never connects: REST commands succeed, state updates never
# arrive. It is listed explicitly here for that reason.
#
# yt-dlp (music videos, the YouTube sign-in) comes with the "ytdlp" plugin, which
# Music requires. The optional real-audio visualiser needs requirements-audio.txt.

httpx>=0.27                 # also used by the Twitch module
python-socketio>=5.11       # Socket.IO client for the companion server feed
python-engineio>=4.9        # Engine.IO v4 transport (pulled in by socketio)
aiohttp>=3.9                # required by socketio.AsyncClient
